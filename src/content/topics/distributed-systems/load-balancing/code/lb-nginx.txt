# nginx as an L7 load balancer in front of three app servers
upstream orders {
    least_conn;                                   # send to the server with fewest active requests
    server 10.0.1.10:8080 weight=2 max_fails=3 fail_timeout=10s;   # bigger box: double share
    server 10.0.1.11:8080 max_fails=3 fail_timeout=10s;            # passive health check
    server 10.0.1.12:8080 max_fails=3 fail_timeout=10s;
    keepalive 64;                                 # reuse upstream connections
}

server {
    listen 443 ssl http2;                         # TLS terminates here
    ssl_certificate     /etc/ssl/orders.crt;
    ssl_certificate_key /etc/ssl/orders.key;

    location /api/ {
        proxy_pass http://orders;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Request-Id $request_id;
        proxy_connect_timeout 1s;
        proxy_read_timeout 10s;
        proxy_next_upstream error timeout http_502 http_503;   # retry on another server…
        proxy_next_upstream_tries 2;                           # …but only once (and only idempotent methods)
    }
}
